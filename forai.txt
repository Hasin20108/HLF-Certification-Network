document.addEventListener('DOMContentLoaded', () => {
    // Form and element references
    const createStudentForm = document.getElementById('create-student-form');
    const addCourseForm = document.getElementById('add-course-form');
    const studentListDiv = document.getElementById('student-list');
    const profilePlaceholder = document.getElementById('profile-placeholder');
    const profileDetailsDiv = document.getElementById('profile-details');
    const resultDiv = document.getElementById('result');
    const searchInput = document.getElementById('search-student-input'); // Search input

    let allStudents = []; // To store the complete list of students

    // --- Helper Functions ---
    const showResult = (data, success = true) => {
        resultDiv.innerHTML = `<pre>${JSON.stringify(data, null, 2)}</pre>`;
        resultDiv.className = success ? 'alert alert-success' : 'alert alert-danger';
    };

    // Renders the list of students based on the provided array
    const renderStudentList = (studentsToRender) => {
        studentListDiv.innerHTML = '';
        if (studentsToRender.length === 0) {
            studentListDiv.innerHTML = '<p class="text-muted p-2">No students match your search.</p>';
            return;
        }
        studentsToRender.forEach(student => {
            const item = document.createElement('a');
            item.href = '#';
            item.className = 'list-group-item list-group-item-action';
            item.innerHTML = `<strong>${student.name}</strong><br><small class="text-muted">${student.studentId}</small>`;
            item.dataset.studentId = student.studentId;
            item.addEventListener('click', (e) => {
                e.preventDefault();
                renderProfileDetails(student.studentId);
            });
            studentListDiv.appendChild(item);
        });
    };
    
    const renderProfileDetails = (studentId) => {
        const student = allStudents.find(s => s.studentId === studentId);
        if (!student) return;

        profilePlaceholder.classList.add('d-none');
        profileDetailsDiv.classList.remove('d-none');

        document.getElementById('profile-name').textContent = student.name;
        document.getElementById('profile-id').textContent = student.studentId;
        document.getElementById('profile-email').textContent = student.email;
        
        const coursesTableBody = document.getElementById('courses-table-body');
        coursesTableBody.innerHTML = `
            <tr>
                <td><strong>Final CGPA</strong></td>
                <td>${student.cgpa}</td>
            </tr>
            <tr>
                <td><strong>Status</strong></td>
                <td><span class="badge bg-info text-dark">${student.status}</span></td>
            </tr>
        `;
        
        document.getElementById('courseStudentId').value = student.studentId;
        const cgpaInput = document.getElementById('cgpa-input');
        cgpaInput.value = student.cgpa !== 'N/A' ? student.cgpa : '';
    };

    // --- API Call Functions ---
    const fetchAllStudents = async () => {
        try {
            const response = await fetch('/getAllStudents'); 
            const result = await response.json();
            if (response.ok) {
                allStudents = result.students || [];
                renderStudentList(allStudents); // Render the full list initially
            } else {
                throw new Error(result.message || 'Failed to fetch students');
            }
        } catch (error) {
            showResult({ error: error.message }, false);
            studentListDiv.innerHTML = `<p class="text-danger p-2">Error loading students.</p>`;
        }
    };

    // --- Event Listeners ---
    createStudentForm.addEventListener('submit', async (e) => {
        e.preventDefault();
        const studentId = document.getElementById('studentId').value;
        const name = document.getElementById('studentName').value;
        const email = document.getElementById('studentEmail').value;
        
        showResult({ message: "Creating new student profile..." });
        try {
            const response = await fetch('/newStudent', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ studentId, name, email }),
            });
            const result = await response.json();
            showResult(result, response.ok);
            if(response.ok) {
                createStudentForm.reset();
                fetchAllStudents(); // Refresh the list
            }
        } catch (error) {
            showResult({ error: error.message }, false);
        }
    });

    addCourseForm.addEventListener('submit', async (e) => {
        e.preventDefault();
        const studentId = document.getElementById('courseStudentId').value;
        const finalCGPA = document.getElementById('cgpa-input').value;
        
        if(!studentId) {
            showResult({ message: 'Please select a student from the list first.'}, false);
            return;
        }

        showResult({ message: `Awarding final CGPA to student ${studentId}...` });
        try {
            const response = await fetch('/awardCGPA', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ studentId, finalCGPA }),
            });
            const result = await response.json();
            showResult(result, response.ok);
            if(response.ok) {
                await fetchAllStudents(); // Refresh data
                setTimeout(() => renderProfileDetails(studentId), 100); // Re-render profile
            }
        } catch (error) {
            showResult({ error: error.message }, false);
        }
    });

    // ## নতুন সার্চের জন্য Event Listener ##
    searchInput.addEventListener('input', (e) => {
        const searchTerm = e.target.value.toLowerCase();
        const filteredStudents = allStudents.filter(student =>
            student.name.toLowerCase().includes(searchTerm) ||
            student.studentId.toLowerCase().includes(searchTerm)
        );
        renderStudentList(filteredStudents);
    });

    // Initial load
    fetchAllStudents();
});

'use strict';

const { Wallets } = require('fabric-network');
const fs = require('fs');
const path = require('path');

// Capture the absolute path to the crypto materials
const crypto_materials = path.resolve(__dirname, '../network/crypto-config');

async function addToWallet() {
    try {
        // Create a new file system based wallet for managing identities.
        const walletPath = path.join(process.cwd(), 'identity/me');
        const wallet = await Wallets.newFileSystemWallet(walletPath);
        console.log(`Wallet path: ${walletPath}`);

        // Check to see if we've already enrolled the admin user.
        const identityExists = await wallet.get('ME_ADMIN');
        if (identityExists) {
            console.log('An identity for the admin user "ME_ADMIN" already exists in the wallet');
            return; // Exit the function if identity is already there
        }

        // Identify credentials for me admin
        const keyPath = path.join(crypto_materials, '/peerOrganizations/me.com/users/Admin@me.com/msp/keystore');
        const certPath = path.join(crypto_materials, '/peerOrganizations/me.com/users/Admin@me.com/msp/signcerts/Admin@me.com-cert.pem');

        // Check if credential files exist
        if (!fs.existsSync(certPath) || !fs.existsSync(keyPath)) {
            throw new Error('Credential files for ME_ADMIN not found!');
        }
        
        // Read credentials
        const cert = fs.readFileSync(certPath, 'utf8');
        // Find the private key file name
        const keyFile = fs.readdirSync(keyPath)[0];
        const key = fs.readFileSync(path.join(keyPath, keyFile), 'utf8');
        
        // Create the identity object for the wallet
        const identity = {
            credentials: {
                certificate: cert,
                privateKey: key,
            },
            mspId: 'meMSP',
            type: 'X.509',
        };
        
        // Import the new identity into the wallet.
        await wallet.put('ME_ADMIN', identity);
        console.log('Successfully imported ME_ADMIN identity into the wallet');

    } catch (error) {
        console.error(`Failed to enroll admin user "ME_ADMIN": ${error.message}`);
        throw error; // Re-throw the error to be caught by the server starter
    }
}

// Export the function so other files can use it
module.exports = { addToWallet };

const fs = require('fs');
const yaml = require('js-yaml');
const { Wallets, Gateway } = require('fabric-network');
let gateway;


async function getContractInstance() {
	
	// A gateway defines which peer is used to access Fabric network
	gateway = new Gateway();
	
	// A wallet is where the credentials to be used for this transaction exist
	const walletPath = './identity/me';
	const wallet = await Wallets.newFileSystemWallet(walletPath);
	
	// What is the username of this Client user accessing the network?
	const fabricUserName = 'ME_ADMIN';
	
	// Load connection profile; will be used to locate a gateway; The CCP is converted from YAML to JSON.
	let connectionProfile = yaml.load(fs.readFileSync('./connection-profile-me.yaml', 'utf8'));
	
	// Set connection options; identity and wallet
	let connectionOptions = {
		wallet: wallet,
		identity: fabricUserName,
		discovery: { enabled: true, asLocalhost: true }
	};
	
	// Connect to gateway using specified parameters
	console.log('.....Connecting to Fabric Gateway');
	await gateway.connect(connectionProfile, connectionOptions);
	
	// Access certification channel
	console.log('.....Connecting to channel - certificationchannel');
	const channel = await gateway.getNetwork('certificationchannel');
	
	// Get instance of deployed Certnet contract
	// The contract name is omitted as there is only one contract in the chaincode
	console.log('.....Connecting to Certnet Smart Contract');
	return channel.getContract('certnet');
}

function disconnect() {
	console.log('.....Disconnecting from Fabric Gateway');
	gateway.disconnect();
}

module.exports.getContractInstance = getContractInstance;
module.exports.disconnect = disconnect;

// Student describes basic details of a student and their academic record

	UpdatedAt  time.Time `json:"updatedAt"`
}

// InitLedger is a dummy function to satisfy the chaincode interface
func (s *SmartContract) InitLedger(ctx contractapi.TransactionContextInterface) error {
	fmt.Println("certnet chaincode initialized")
	return nil
}

// CreateStudent creates a new student profile on the ledger.
func (s *SmartContract) CreateStudent(ctx contractapi.TransactionContextInterface, studentID string, name string, email string) (*Student, error) {
	existing, err := ctx.GetStub().GetState(studentID)
	if err != nil {
		return nil, fmt.Errorf("failed to read from world state: %v", err)
	}
	if existing != nil {
		return nil, fmt.Errorf("the student %s already exists", studentID)
	}

	clientIdentity, err := ctx.GetClientIdentity().GetMSPID() // Get the MSP ID of the caller (e.g., "ruMSP")
	if err != nil {
		return nil, fmt.Errorf("failed to get client MSPID: %v", err)

// GetAllStudents returns all students found in the world state.
func (s *SmartContract) GetAllStudents(ctx contractapi.TransactionContextInterface) ([]*Student, error) {
	// Range query with empty startKey and endKey does not work.
	// We need to use a query selector to get all students.
	queryString := `{"selector":{"docType":"student"}}`
	resultsIterator, err := ctx.GetStub().GetQueryResult(queryString)
	if err != nil {
		return nil, err
	}
	defer resultsIterator.Close()

	var students []*Student
	for resultsIterator.HasNext() {
		queryResponse, err := resultsIterator.Next()
		if err != nil {
			return nil, err
		}

		var student Student
		err = json.Unmarshal(queryResponse.Value, &student)
		if err != nil {
			return nil, err
		}
		students = append(students, &student)
	}

	return students, nil
}
